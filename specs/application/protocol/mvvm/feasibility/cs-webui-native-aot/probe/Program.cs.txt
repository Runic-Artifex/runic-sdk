// Copy this transcript to Program.cs before reproducing the temporary probe.
using CsWebUi;
using CsWebUi.Native;

namespace WebUIToolkit.MVVM.UpstreamAotProbe;

internal static class Program
{
    private static readonly ManualResetEventSlim CallbackReceived = new(false);

    private static int Main()
    {
        var libraryPath = Environment.GetEnvironmentVariable("CSWEBUI_NATIVE_LIBRARY");
        if (string.IsNullOrWhiteSpace(libraryPath))
        {
            Console.Error.WriteLine("CSWEBUI_NATIVE_LIBRARY is required.");
            return 2;
        }

        WebUiNativeLibrary.SetLibraryPath(libraryPath);
        WebUiApplication.SetConnectionTimeout(10);

        using var window = new WebUiWindow();
        using var binding = window.Bind("roundTrip", static webUiEvent =>
        {
            var value = webUiEvent.GetString();
            Console.WriteLine($"CALLBACK argument={value} client={webUiEvent.ClientId}");
            CallbackReceived.Set();
            return WebUiResult.FromString($"ack:{value}");
        });

        var url = window.StartServer("""
            <!doctype html>
            <html lang="en">
            <head><meta charset="utf-8"><script src="/webui.js"></script></head>
            <body>pending
            <script>
            addEventListener('DOMContentLoaded', async () => {
              while (!globalThis.webui?.isConnected()) {
                await new Promise(resolve => setTimeout(resolve, 25));
              }
              document.body.textContent = await globalThis.webui.call('roundTrip', 'native-aot');
            });
            </script>
            </body></html>
            """);

        Console.WriteLine($"READY {url}");
        Console.Out.Flush();

        if (!CallbackReceived.Wait(TimeSpan.FromSeconds(30)))
        {
            Console.Error.WriteLine("Timed out waiting for the browser-to-managed callback.");
            return 3;
        }

        window.Close();
        WebUiApplication.Clean();
        Console.WriteLine("PASS browser-to-managed-to-browser callback");
        return 0;
    }
}
